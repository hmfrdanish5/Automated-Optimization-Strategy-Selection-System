# Hand-written TAC programs for the first prototype.
# A line of the form === program_id === starts a program.
# These texts are input only. best_strategy is not assigned here.

=== constant_heavy ===
# Three binary expressions can be evaluated immediately.
t1 = 10 + 20
t2 = 5 * 4
t3 = 100 / 5
t4 = t1 + t2
print t3
print t4

=== propagation_ready ===
# Constants are copied into later expressions, but nothing can fold yet.
x = 10
y = x + 5
z = y * 2
print z

=== algebraic_heavy ===
t1 = a + 0
t2 = b - 0
t3 = c * 1
t4 = d / 1
t5 = e * 0
t6 = 0 + f
print t1
print t2
print t3
print t4
print t5
print t6

=== cse_heavy ===
# a + b is computed three times, so there are two repeated copies.
t1 = a + b
t2 = c * d
t3 = a + b
t4 = a + b
print t1
print t2
print t3
print t4

=== dead_code_heavy ===
# x and z feed nothing observable. y is printed, so y = 20 stays live.
x = 10
y = 20
z = x + 1
print y

=== loop_heavy ===
i = 0
n = 10
L1:
t1 = i * 2
i = i + 1
if i < n goto L1
print t1

=== mixed ===
t1 = 10 + 20
t2 = a + 0
t3 = a + b
t4 = a + b
unused = t1 + 1
print t2
print t4

=== straight_line ===
t1 = a + b
t2 = t1 * c
print t2

=== expression_redefined ===
# The text of a + b repeats, but a is assigned in between.
# The feature count does not decide whether CSE is safe.
t1 = a + b
a = 4
t2 = a + b
print t1
print t2
